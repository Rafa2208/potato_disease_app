import streamlit as st
import torch
import torchvision.transforms as transforms
from torchvision import models
from PIL import Image
import joblib

# Load Random Forest model
rf = joblib.load("random_forest_model.pkl")
class_names = ['Early_Blight', 'Late_Blight', 'Healthy']  # Update if different

# Load ResNet feature extractor
resnet = models.resnet50(pretrained=True)
resnet.fc = torch.nn.Identity()
resnet.load_state_dict(torch.load("resnet_feature_extractor.pth", map_location=torch.device('cpu')))
resnet.eval()

# Preprocessing
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406],
                         [0.229, 0.224, 0.225])
])

# Streamlit UI
st.title("Potato Disease Detection App 🌿")
st.write("Upload a potato leaf image and get the disease prediction.")

uploaded_file = st.file_uploader("Choose a leaf image...", type=["jpg", "jpeg", "png"])
if uploaded_file is not None:
    image = Image.open(uploaded_file).convert("RGB")
    st.image(image, caption='Uploaded Image', use_column_width=True)

    img_t = transform(image).unsqueeze(0)
    with torch.no_grad():
        features = resnet(img_t).numpy()

    prediction = rf.predict(features)[0]
    st.success(f"Prediction: **{class_names[prediction]}**")
